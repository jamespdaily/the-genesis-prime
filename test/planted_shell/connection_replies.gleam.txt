import genesis_prime/connection/shell
import genesis_prime/step.{CallerId, Effects, OwnEffect, Reply, Rule}
import genesis_prime/connection/script
import gleam/option.{None}

/// A connection rule that answers a caller, handed to the connection shell.
pub fn settings() {
  shell.Settings(
    rule: Rule(
      name: "planted",
      settings: Nil,
      new: fn(_settings) { Nil },
      step: fn(state, _input) {
        #(state, Effects([OwnEffect(Reply(CallerId(1), Nil))]))
      },
    ),
    send_deadline_ms: 100,
    shutdown_ms: 100,
    wire_recorder: None,
    scripts: script.operating_system,
    directory: fn(_request, _deadline, _reply) { Error(Nil) },
    registry: None,
    locate: fn(_address) { Error(Nil) },
    deliver: fn(_address, _request) { Error(Nil) },
  )
}
